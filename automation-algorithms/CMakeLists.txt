cmake_minimum_required(VERSION 3.20)
project(automation_algorithms C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(aa STATIC src/debounce.c src/ramp.c src/hsm.c src/pid.c src/filters.c)
target_include_directories(aa PUBLIC include)
target_link_libraries(aa PUBLIC m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(aa PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wdouble-promotion -Werror)
endif()

option(AA_BUILD_TESTS "Pruebas y ejemplos (en el PC)" ON)
if(AA_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
  option(AA_SANITIZE "AddressSanitizer + UBSan" OFF)
  if(AA_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
    add_link_options(-fsanitize=address,undefined)
  endif()
  include(FetchContent)
  FetchContent_Declare(unity GIT_REPOSITORY https://github.com/ThrowTheSwitch/Unity.git GIT_TAG v2.6.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(unity)
  enable_testing()
  add_executable(test_aa tests/test_aa.c)
  target_link_libraries(test_aa PRIVATE aa unity)
  target_compile_definitions(unity PUBLIC UNITY_INCLUDE_FLOAT UNITY_INCLUDE_DOUBLE)
  add_test(NAME aa COMMAND test_aa)

  foreach(ex pid_windup stepper_profile debounce_demo hsm_traffic_light)
    add_executable(${ex} examples/${ex}.c)
    target_link_libraries(${ex} PRIVATE aa)
  endforeach()
endif()
